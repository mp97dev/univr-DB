db.museo.updateMany( {citta: 'Verona'}, {$set: {prezzoRidotto: 5}});

db.museo.find( {'mostre.inizio': {$lte: ISODate('2025-05-26')}, 'mostre.fine': {$gte: ISODate('2025-05-26')}}, {nome: 1, 'mostre.titolo': 1} );

db.museo_ref.aggregate(
[
{$lookup:{ from: 'mostra', localField: 'mostre', foreignField: '_id', as: 'ope'}},
